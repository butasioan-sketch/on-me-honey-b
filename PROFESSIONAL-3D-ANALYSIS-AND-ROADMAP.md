## 🔧 TECHNISCHE IMPLEMENTIERUNG

### THREE.JS SETUP (PROFESSIONELL)

```typescript
// Enhanced 3D Viewer Setup
import { Canvas } from '@react-three/fiber';
import { 
  OrbitControls, 
  Environment, 
  PerspectiveCamera,
  ContactShadows,
  Stage,
  Lightformer,
  useGLTF,
  useTexture
} from '@react-three/drei';
import { 
  EffectComposer, 
  Bloom, 
  ToneMapping,
  SSAO 
} from '@react-three/postprocessing';

function Professional3DViewer({ modelPath, designTexture }) {
  return (
    <Canvas
      dpr={[1, 2]} // Retina support
      gl={{ 
        antialias: true, 
        alpha: true,
        powerPreference: 'high-performance'
      }}
      shadows
    >
      {/* Professional Lighting Setup */}
      <Environment preset="studio" />
      
      {/* Additional Studio Lights */}
      <ambientLight intensity={0.3} />
      <directionalLight 
        position={[10, 10, 5]} 
        intensity={1}
        castShadow
        shadow-mapSize={[2048, 2048]}
      />
      <directionalLight 
        position={[-10, 10, -5]} 
        intensity={0.5}
      />
      
      {/* Product Model */}
      <ProductModel 
        modelPath={modelPath}
        designTexture={designTexture}
      />
      
      {/* Ground Shadows */}
      <ContactShadows 
        position={[0, -1, 0]}
        opacity={0.4}
        scale={10}
        blur={2}
      />
      
      {/* Interactive Controls */}
      <OrbitControls
        enableZoom={true}
        enablePan={false}
        minDistance={2}
        maxDistance={10}
        autoRotate={true}
        autoRotateSpeed={1}
      />
      
      {/* Post-Processing Effects */}
      <EffectComposer>
        <Bloom 
          intensity={0.3} 
          luminanceThreshold={0.9}
        />
        <ToneMapping />
        <SSAO />
      </EffectComposer>
    </Canvas>
  );
}

// PBR Material Setup
function ProductModel({ modelPath, designTexture }) {
  const { scene } = useGLTF(modelPath);
  const texture = useTexture(designTexture);
  
  // Apply PBR material
  scene.traverse((child) => {
    if (child.isMesh) {
      child.material = new THREE.MeshStandardMaterial({
        map: texture,
        normalMap: normalMap,
        roughnessMap: roughnessMap,
        aoMap: aoMap,
        envMapIntensity: 1,
        metalness: 0,
        roughness: 0.7,
      });
      child.castShadow = true;
      child.receiveShadow = true;
    }
  });
  
  return <primitive object={scene} />;
}
```
